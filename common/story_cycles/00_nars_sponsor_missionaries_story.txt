story_sponsor_meithios = {
    visible = yes
    icon = { reference = "gfx/interface/icons/artifact/artifact_chronicle.dds" }
    background = { reference = "gfx/interface/illustrations/event_scenes/nars_temple.dds" }

    visualization = {
        basic_counter = {
            variable_name = "sponsor_meithios_success"
            tooltip = "SPONSOR_MISSIONARIES_COUNTER_TOOLTIP"
            min = 0
            max = 3
            label = "SPONSOR_MISSIONARIES_VALUE_LABEL"
            min_label = "SPONSOR_MISSIONARIES_VALUE_MIN_LABEL"
            max_label = "SPONSOR_MISSIONARIES_VALUE_MAX_LABEL"
        }
    }

    on_setup = {
        set_variable = { name = sponsor_meithios_success value = 0 }
    }

    effect_group = {
        days = 5
        trigger = {
            exists = story_owner
        }
        triggered_effect = {
            trigger = {
                NOT = {
                    story_owner = {
                        any_scheme = {
                            type = sponsor_meithios_scheme
                        }
                    }
                }
            }
            effect = {
                end_story = yes
            }
        }
    }

    on_owner_death = { end_story = yes }
    on_end = {
        debug_log = "$decision_sponsor_meithios$ story cycle ended on:"
        debug_log_date = yes
    }
}
