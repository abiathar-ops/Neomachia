selian_vassal_conversion_effect = {
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		random = {
			chance = 65
			set_character_faith = root.faith
			set_character_rite = root.rite
		}
	}
	every_vassal = {
		limit = {
			liege = root
			is_ai = yes
		}
		random = {
			chance = 65
			set_culture = culture:selian
		}
	}
}

selian_settle_county_effect = {
    # Save the character triggering the effect so we can reference them safely
    save_scope_as = selian_ruler
    
    every_sub_realm_county = {
        limit = {
            # Check if the county belongs to a Duchy held by the ruler, 
            # OR if it belongs to the de jure Duchy of the ruler's capital.
            duchy = {
                OR = {
                    holder = scope:selian_ruler
                    this = scope:selian_ruler.capital_county.duchy
                }
            }
            
            # Only target the county if it actually needs a conversion
            OR = {
                NOT = { faith = root.faith }
                NOT = { culture = culture:selian }
            }
        }
        
        # Convert Faith if needed
        if = {
            limit = {
                NOT = { faith = root.faith }
            }
            set_county_faith = root.faith
            set_county_rite = root.rite
        }
        
        # Convert Culture if needed
        if = {
            limit = {
                NOT = { culture = culture:selian }
            }
            set_county_culture = culture:selian
        }
        
        # Apply the control loss once
        change_county_control = major_county_control_loss
    }
}

selian_diplomacy_entreaty_effect = {
    custom_tooltip = selian_diplomacy.tt
    
    hidden_effect = {
        # Dynamically identify the remaining sibling based on the root character's modifier
        if = {
            limit = { has_character_modifier = eleios_modifier }
            father = {
                random_child = { limit = { has_character_modifier = ilione_modifier } save_scope_as = target_1 }
            }
        }
        else_if = {
            limit = { has_character_modifier = ilione_modifier }
            father = {
                random_child = { limit = { has_character_modifier = eleios_modifier } save_scope_as = target_1 }
            }
        }
        
        random_list = {
            # Success: Target 1 joins your court along with their followers
            50 = {
                modifier = {
                    add = 25
                    scope:target_1 = { opinion = { target = root value >= 50 } }
                }
                
                send_interface_message = {
                    title = selian_diplomacy_joins_toast_t
                    desc = selian_diplomacy_joins_toast_tt
                    type = event_diplomacy_good_with_text
                    left_icon = scope:target_1
                }
                
                scope:target_1 = {
                    hidden_effect = { laamp_clear_domicile_buildings_liquidator_effect = yes }
                    every_courtier = { add_to_list = target_1_followers_list }
                    destroy_laamp_effect = { ADVENTURER = this }
                    remove_trait = adventurer
                    
                    root = { 
                        add_courtier = scope:target_1 
                        return_to_court = yes 
                    }
                    
                    hidden_effect = {
                        every_in_list = {
                            list = target_1_followers_list
                            limit = { NOT = { liege_or_court_owner = root } }
                            root = { add_courtier = prev }
                            remove_trait = adventurer_follower
                        }
                    }
                }
            }
            
            # Failure: Target 1 refuses, gains disapproving opinion, and becomes a rival
            50 = {
                send_interface_message = {
                    title = selian_diplomacy_refuses_toast_t
                    type = event_diplomacy_bad_with_text
                    left_icon = scope:target_1
                    desc = selian_diplomacy_refuses_toast_tt
                }
                
                scope:target_1 = {
                    add_opinion = {
                        target = root
                        modifier = disapproving_opinion
                    }
                    set_relation_rival = {
                        reason = rival_claimant_fight
                        target = root
                    } 
                }
            }
        }
    }
}
