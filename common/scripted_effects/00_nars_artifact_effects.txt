# Create effects for the holy relics of the Politeion. Call with the future owner as scope (like create_ilyanil_effect).

create_aidokyne_effect = {
	if = {
		limit = {
			has_dlc_feature = royal_court
		}
		create_artifact = {
			name = aidokyne_name
			description = aidokyne_desc
			modifier = artifact_aidokyne_modifier
			template = aidokyne_template
			type = helmet
			visuals = pedestal_justinian

			owner = root
			wealth = 3000
			quality = 100

			save_scope_as = newly_created_artifact
			history = {
				type = created_before_history
			}
		}

		scope:newly_created_artifact = {
			set_artifact_rarity_illustrious = yes
			set_variable = { name = historical_unique_artifact value = yes }
		}
	}
}
